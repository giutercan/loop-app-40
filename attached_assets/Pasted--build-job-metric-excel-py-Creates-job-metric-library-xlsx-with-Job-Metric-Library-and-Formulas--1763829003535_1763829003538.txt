# Build dataframe and write to Excel
df = pd.DataFrame(rows)
xlsx_file = "job_metric_library.xlsx"
with pd.ExcelWriter(xlsx_file, engine="openpyxl") as writer:
    df.to_excel(writer, index=False, sheet_name="Job_Metric_Library")

# Create Formulas & Examples sheet
example_rows = [
    ["Description", "Value"],
    ["QoH delta (index points)", 10],
    ["Hires per year", 120],
    ["Avg margin per hire", 100000],
    ["Realisation rate", 0.6],
    ["Implementation cost", 150000],
    ["Ongoing annual cost", 50000],
    ["Discount rate", 0.08],
    ["Horizon years", 3],
    # Year 1 incremental (formula text)
    ["Year1 incremental (calc)", "=B2*B3*B4*B5"]
]

# Write the example sheet
wb = load_workbook(xlsx_file)
ws = wb.create_sheet("Formulas & Examples")
for r_idx, row in enumerate(example_rows, start=1):
    for c_idx, val in enumerate(row, start=1):
        ws.cell(row=r_idx, column=c_idx, value=val)

# Format headers on Job_Metric_Library
ws_dm = wb["Job_Metric_Library"]
header = ws_dm[1]
for cell in header:
    cell.font = Font(bold=True, color="FFFFFF")
    cell.fill = PatternFill("solid", fgColor="00634F")
    cell.alignment = Alignment(horizontal="center", vertical="center")

# Adjust column widths
for i, col in enumerate(ws_dm.columns, start=1):
    max_length = 0
    for cell in col:
        try:
            if cell.value:
                max_length = max(max_length, len(str(cell.value)))
        except:
            pass
    adjusted_width = (max_length + 2)
    ws_dm.column_dimensions[get_column_letter(i)].width = min(adjusted_width, 60)

# Save workbook
wb.save(xlsx_file)
print(f"Created {xlsx_file} in the current directory.")
